\section{What are We Trying to Accomplish?}

\begin{frame}
\frametitle{What are we trying to accomplish?}
\begin{enumerate}
    \item{
            We want to create digital renders of black holes
    }
    \item{
            We know the equations that define black hole geometry
    }
    \item{
            We might be able to utilize these equations to create scene renders under black hole geometry.
    }
\end{enumerate}
\end{frame}

\section{What Are the Equations?}

\begin{frame}
\frametitle{How to Black Holes Deform Light?}
\begin{enumerate}
    \item{
            The Schwarzschild Metric tells us the paths light rays take in the orbit of a non-spinning neutrally charged black hole
    }
    \item{
            Given M, the mass of the black hole,
            we can simplify the Schwarzschild Metric to obtain $u^{\prime\prime}-u=3Mu^{2}$\cite{gravitation-mtw}. Where $u=\frac1{r}$, 
             with $r$ being the distance between the light ray and the black hole origin,
             and $u^\prime=\frac{du}{d\phi}$, 
             where $\phi$ is the angle in spherical coordinates to the origin of the black hole.
    }
    \item{
            This means that given some initial positional parameters, we are able to trace the path of a light ray as it orbits a black hole using GSL.
    }
\end{enumerate}
\end{frame}

\section{How We Can Use These Equations}
\begin{frame}
\frametitle{Raytracing}
\begin{enumerate}
    \item{
        Since we have a way to trace the path of light rays around a given black hole, we can utilize Raytracing in order to render an image
    }
    \item{
        Raytracers normally compute trajectories under the assumption that light takes a straight path, however, we will need to modify the Raytracer to account for the distortion of black hole geometry.
    }
    \item{
        This technique is called Ray Marching
    }
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Raytracing}
\begin{enumerate}
    \item{
        Raytracers create an image by tracing the path light rays take from
        the aperature of a simulated camera, through the “image”
    }
    \item{
        We use linear algebra to compute collision with various objects in the
        scene.
    }
    \item{
        With proper setup, this can let us create photorealistic images.
    }
\end{enumerate}
\includegraphics[scale=.10]{figs/raytrace.png}
\cite{wikipedia_raytrace}
\end{frame}

\begin{frame}
\frametitle{Speed Challenges}
\begin{enumerate}
    \item{
        Tracing a single ray can require ~2000 computed points for a low quality render (upwards of 40,000 for high quality renders)
    }
    \item{
        A normal sized image can contain 2073600 pixels
    }
    \item{
        Each pixel can require 10 computed rays for anti-aliasing
    }
    \item{
        All of this can be extrordinarily expensive!!
    }
\end{enumerate}
\end{frame}


\begin{frame}
\frametitle{MPI+OpenMP}
\begin{enumerate}
    \item{
        Each pixel can be rendered independently, which means we can break up this image any way we please. 
    }
    \item{
        We break up the image horizontally, and send a different band to each process
    }
    \item{
        Within each band, we can render each scanline on a single OpenMP thread.
    }
\end{enumerate}
\end{frame}
